    pub fn is_cancellable(&self) -> bool {
        !matches!(
            self,

                | CommandSpec::GitPull
                | CommandSpec::GitPush
                | CommandSpec::GitRebase { .. }
                | CommandSpec::GitCheckout { .. }
                | CommandSpec::GitCheckoutNew { .. }
                | CommandSpec::GitFetch
        )
    }
