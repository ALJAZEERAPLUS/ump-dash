    //!
    //! The port's output type is `CommandEvent`, not `Action`. Consumers (effect_runner
    //! after Plan 13-08, or src/app.rs dispatch_command in the interim) translate
    //! CommandEvent → Action at the app-layer boundary.

    use crate::domain::command::CommandSpec;
    use std::path::PathBuf;
    use std::process::ExitStatus;

    #[derive(Debug)]
    pub enum CommandEvent {
        OutputLine(String),
        Exited(ExitStatus),
    }

    /// Spawns a command and streams its lifecycle as typed events.
    pub trait CommandRunnerPort: Send + Sync {
        /// Spawn the command. Returns a receiver that emits each stdout/stderr line
        /// (as OutputLine) in order, followed by exactly one Exited(status) event,
        /// after which the receiver closes. kill_on_drop semantics apply — dropping
        /// the underlying task terminates the subprocess tree.
        fn spawn(
            &self,
            spec: CommandSpec,
            cwd: PathBuf,
            branch: String,
        ) -> tokio::sync::mpsc::UnboundedReceiver<CommandEvent>;
    }

    //! Tokio-backed command runner (F-101 adapter).
    //!
    //! Spawns OS commands via tokio::process::Command with process-group isolation
    //! (`process_group(0)`) and kill_on_drop safety. Emits `CommandEvent`s — never
    //! `Action` — keeping infra ignorant of domain messaging types.

    use crate::domain::command::CommandSpec;
    use crate::domain::ports::command_runner_port::{CommandEvent, CommandRunnerPort};
    use std::path::PathBuf;
    use std::process::Stdio;
    use tokio::io::{AsyncBufReadExt, BufReader};
    use tokio::process::Command;
    use tokio::sync::mpsc;

    pub struct TokioCommandRunner;

    impl CommandRunnerPort for TokioCommandRunner {
        fn spawn(
            &self,
            spec: CommandSpec,
            cwd: PathBuf,
            branch: String,
        ) -> mpsc::UnboundedReceiver<CommandEvent> {
            let (tx, rx) = mpsc::unbounded_channel();
            tokio::spawn(async move {
                run(spec, cwd, branch, tx).await;
            });
            rx
        }
    }

    async fn run(
        spec: CommandSpec,
        cwd: PathBuf,
        branch: String,
        tx: mpsc::UnboundedSender<CommandEvent>,
    ) {
        let argv = build_argv(&spec, &branch);
        if argv.is_empty() {
            let status = std::process::ExitStatus::from_raw(1);  // Unix-only; use .code() on other platforms
            let _ = tx.send(CommandEvent::Exited(status));
            return;
        }
        let mut cmd = Command::new(&argv[0]);
        cmd.args(&argv[1..])
            .current_dir(&cwd)
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .kill_on_drop(true);
        #[cfg(unix)]
        {
            use std::os::unix::process::CommandExt;
            cmd.process_group(0);
        }
        let mut child = match cmd.spawn() {
            Ok(c) => c,
            Err(e) => {
                let _ = tx.send(CommandEvent::OutputLine(format!("spawn error: {e}")));
                // Best-effort exit event
                #[cfg(unix)]
                let status = std::os::unix::process::ExitStatusExt::from_raw(1 << 8);
                #[cfg(not(unix))]
                let status = std::process::ExitStatus::default();
                let _ = tx.send(CommandEvent::Exited(status));
                return;
            }
        };
        let stdout = child.stdout.take();
        let stderr = child.stderr.take();

        let tx_out = tx.clone();
        if let Some(out) = stdout {
            tokio::spawn(async move {
                let mut reader = BufReader::new(out).lines();
                while let Ok(Some(line)) = reader.next_line().await {
                    let _ = tx_out.send(CommandEvent::OutputLine(line));
                }
            });
        }
        let tx_err = tx.clone();
        if let Some(err) = stderr {
            tokio::spawn(async move {
                let mut reader = BufReader::new(err).lines();
                while let Ok(Some(line)) = reader.next_line().await {
                    let _ = tx_err.send(CommandEvent::OutputLine(line));
                }
            });
        }

        match child.wait().await {
            Ok(status) => { let _ = tx.send(CommandEvent::Exited(status)); }
            Err(_e) => {
                #[cfg(unix)]
                let status = std::os::unix::process::ExitStatusExt::from_raw(1 << 8);
                #[cfg(not(unix))]
                let status = std::process::ExitStatus::default();
                let _ = tx.send(CommandEvent::Exited(status));
            }
        }
    }
