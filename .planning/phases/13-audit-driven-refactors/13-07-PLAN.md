### F-201 consumer — update.rs rewrite

Target signature per AUDIT F-201 line 417:
```rust
pub fn update(state: &mut AppState, action: Action) -> Vec<Effect>
```

Transformation per 13-PATTERNS.md:809-849. Every arm that currently calls `tokio::spawn`
pushes an Effect variant instead. Example:
```rust
// BEFORE:
Action::MetroStart => {
    state.palette_mode = None;
    if state.metro.is_running() {
        state.pending_restart = true;
        update(state, Action::MetroStop, metro_tx, handle_tx);
        return;
    }
    if state.skip_external_metro_check {
        state.skip_external_metro_check = false;
        let _ = metro_tx.send(Action::MetroStartConfirmed);
        return;
    }
    let tx = metro_tx.clone();
    tokio::spawn(async move {
        if let Some(info) = crate::infra::port::detect_external_metro(8081).await {
            let _ = tx.send(Action::ExternalMetroDetected(info.into()));
        } else {
            let _ = tx.send(Action::MetroStartConfirmed);
        }
    });
}

// AFTER (update.rs in pure form):
Action::MetroStart => {
    state.palette_mode = None;
    let mut effects = Vec::new();
    if state.metro.is_running() {
        state.pending_restart = true;
        effects.extend(update(state, Action::MetroStop));
        return effects;
    }
    if state.skip_external_metro_check {
        state.skip_external_metro_check = false;
        effects.push(Effect::ScheduleAction(Action::MetroStartConfirmed));
        return effects;
    }
    effects.push(Effect::DetectExternalMetro { port: 8081 });
    effects
}
```

Recursive `update()` self-dispatch (F-206): every recursive call becomes either
`effects.extend(update(state, Action::X))` (inline continuation) OR
`effects.push(Effect::ScheduleAction(Action::X))` (deferred via the action queue). See
13-RESEARCH.md Pitfall 3 — `pending_restart` STAYS (metro-lifecycle, not prereq);
`pending_metro_run`/`pending_metro_after_sync` are candidates for Recipe variants in
Plan 13-09 but stay alive in THIS plan.

**Effect enum additions during this plan (if needed):**
The base Effect enum from Plan 13-03 had ScheduleAction. If specific update() arms
need a variant Plan 13-03 didn't sketch, ADD it to src/app/effect.rs with a comment
pointing to the app.rs line that motivates it.

pub fn handle_key(state: &AppState, key: KeyEvent) -> Option<Action> {
    if key.kind != KeyEventKind::Press { return None; }
    for kb in KEYBINDINGS.iter() {

```rust
// src/app/effect_runner.rs (partial — full impl in Plan 13-08)
use crate::app::effect::Effect;
use crate::domain::action::Action;
use crate::domain::ports::metro_port::MetroPort;
use std::sync::Arc;
use tokio::sync::mpsc::UnboundedSender;

pub struct EffectRunner {
    pub metro: Arc<dyn MetroPort>,
    pub action_tx: UnboundedSender<Action>,
    // Other adapter fields added in Plan 13-08
}

impl EffectRunner {
    pub async fn run_effects(&self, effects: Vec<Effect>) {
        for effect in effects {
            match effect {
                Effect::ScheduleAction(action) => {
                    let _ = self.action_tx.send(action);
                }
                Effect::DetectExternalMetro { port: _ } => {
                    // TODO(Plan 13-08): call self.port_probe.detect_external(port)
                    // For NOW: spawn inline like the current code does
                    let tx = self.action_tx.clone();
                    tokio::spawn(async move {
                        match crate::infra::port::detect_external_metro(8081).await {
                            Some(info) => { let _ = tx.send(Action::ExternalMetroDetected(crate::domain::ports::port_probe_port::ExternalProcessInfo { pid: info.pid, working_dir: info.working_dir })); }
                            None => { let _ = tx.send(Action::MetroStartConfirmed); }
                        }
                    });
                }
                Effect::SpawnMetro { worktree } => {
                    let metro = self.metro.clone();
                    let tx = self.action_tx.clone();
                    let activity_tx = tx.clone();
                    tokio::spawn(async move {
                        let on_activity: Box<dyn Fn(MetroActivity) + Send + Sync> = Box::new(move |act| {
                            let _ = activity_tx.send(Action::MetroActivityUpdate(act));
                        });
                        match metro.start(worktree, on_activity).await {
                            Ok(handle) => { /* handle registration flows back via another Action */ }
                            Err(e) => { let _ = tx.send(Action::MetroStartFailed(e.to_string())); }
                        }
                    });
                }
                Effect::MetroHttpPost { url, body } => {
                    let metro = self.metro.clone();
                    tokio::spawn(async move {
                        let _ = metro.http_post(&url, &body).await;
                    });
                }
                Effect::KillProcess { pid } => {
                    tokio::spawn(async move {
                        let _ = crate::infra::port::kill_process(pid).await;
                    });
                }
                // Other variants — Plan 13-08 implements these via the Adapters struct
                _ => {
                    tracing::warn!("Effect variant not yet handled by effect_runner (Plan 13-08 TODO): {:?}", effect);
                }
            }
        }
    }
}
```

                && !state.show_help

                && state.modal.is_none()

            BindingContext::CommandOutput => state.focused_panel == FocusedPanel::CommandOutput

            _ => false,
